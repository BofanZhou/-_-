二、五段弹药（Python，离线可跑）

① 最小二乘/非线性拟合 + R²（R² 进摘要）

import numpy as np
from scipy.optimize import curve_fit

x = np.array([...]); y = np.array([...])          # 你的数据
f = lambda x, a, b, c: a*x**2 + b*x + c           # 换成你的模型形式
(a,b,c), _ = curve_fit(f, x, y)
yhat = f(x, a, b, c)
r2 = 1 - ((y-yhat)**2).sum() / ((y-y.mean())**2).sum()
print(a, b, c, "R2 =", round(r2, 4))

③ PSO 骨架（30 行，当主算法或复证算法都行）

import numpy as np
def pso(f, dim, lo, hi, n=40, iters=200, w=0.7, c1=1.5, c2=1.5):
    X = np.random.uniform(lo, hi, (n, dim)); V = np.zeros_like(X)
    p = X.copy(); pf = np.array([f(x) for x in X])      # 个体最优
    g = p[pf.argmin()].copy(); gf = pf.min()             # 全局最优（最小化）
    for _ in range(iters):
        r1, r2 = np.random.rand(2)
        V = w*V + c1*r1*(p-X) + c2*r2*(g-X)
        X = np.clip(X+V, lo, hi)
        fit = np.array([f(x) for x in X])
        m = fit < pf; p[m], pf[m] = X[m], fit[m]
        if pf.min() < gf: g, gf = p[pf.argmin()].copy(), pf.min()
    return g, gf

g, best = pso(lambda z: ((z-3)**2).sum(), dim=5, lo=-10, hi=10)
